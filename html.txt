    <script>
        // DEMO ADMIN CREDENTIALS
        const ADMIN_CREDENTIALS = {
            username: "YUVA001",
            password: "Admin@123"
        };

        // APPLICATION STATE
        let currentUser = null;
        let isRegistering = false;
        let generatedOtp = null;
        let isOtpVerified = false;

        // LOCAL STORAGE
        function getUsers() {
            return JSON.parse(localStorage.getItem("ry_users") || "[]");
        }

        function saveUsers(users) {
            localStorage.setItem("ry_users", JSON.stringify(users));
        }

        function getLogs() {
            return JSON.parse(localStorage.getItem("ry_logs") || "[]");
        }

        function saveLogs(logs) {
            localStorage.setItem("ry_logs", JSON.stringify(logs));
        }

        // NOTIFICATIONS
        function showToast(message, type = "info") {
            const container = document.getElementById("toastContainer");
            const toast = document.createElement("div");

            const colors = {
                success: "bg-emerald-600",
                error: "bg-red-600",
                info: "bg-slate-800"
            };

            toast.className = `toast ${colors[type] || colors.info} text-white px-5 py-3 rounded-xl shadow-lg`;
            toast.textContent = message;

            container.appendChild(toast);

            setTimeout(() => toast.remove(), 3000);
        }

        // INITIALIZE
        window.addEventListener("DOMContentLoaded", () => {
            const session = sessionStorage.getItem("ry_session");

            if (session) {
                currentUser = JSON.parse(session);
                updateNav();

                if (currentUser.isAdmin) {
                    showAdminDashboard();
                } else {
                    showUserDashboard();
                }
            } else {
                showLanding();
            }
        });

        // NAVIGATION
        function hideAllSections() {
            document.getElementById("landingSection").classList.add("hidden");
            document.getElementById("userDashboard").classList.add("hidden");
            document.getElementById("adminDashboard").classList.add("hidden");
        }

        function showLanding() {
            hideAllSections();
            document.getElementById("landingSection").classList.remove("hidden");
        }

        function showUserDashboard() {
            if (!currentUser || currentUser.isAdmin) return;

            hideAllSections();
            document.getElementById("userDashboard").classList.remove("hidden");

            renderUserProfile();
            renderUserLogs();
        }

        function showAdminDashboard() {
            if (!currentUser || !currentUser.isAdmin) return;

            hideAllSections();
            document.getElementById("adminDashboard").classList.remove("hidden");

            renderAdminReports();
            renderAdminStaff();
        }

        function updateNav() {
            const loginBtn = document.getElementById("loginNavBtn");
            const badge = document.getElementById("userInfoBadge");

            if (currentUser) {
                loginBtn.classList.add("hidden");
                badge.classList.remove("hidden");
                badge.classList.add("flex");

                document.getElementById("navUserName").textContent =
                    currentUser.isAdmin ? "Admin Manager" : currentUser.name;

                document.getElementById("navUserRole").textContent =
                    currentUser.isAdmin ? "Administrator" : currentUser.role;
            } else {
                loginBtn.classList.remove("hidden");
                badge.classList.add("hidden");
                badge.classList.remove("flex");
            }
        }

        function logout() {
            currentUser = null;
            sessionStorage.removeItem("ry_session");

            updateNav();
            showLanding();
            showToast("Logged out successfully.");
        }

        // AUTHENTICATION MODAL
        function showAuthModal(tab = "user") {
            document.getElementById("authModal").classList.remove("hidden");
            switchAuthTab(tab);
        }

        function closeAuthModal() {
            document.getElementById("authModal").classList.add("hidden");
        }

        function switchAuthTab(tab) {
            const isUser = tab === "user";

            document.getElementById("userAuthContainer").classList.toggle("hidden", !isUser);
            document.getElementById("adminAuthContainer").classList.toggle("hidden", isUser);

            document.getElementById("tabUserBtn").className = isUser
                ? "flex-1 bg-blue-600 text-white p-3 rounded-lg font-bold"
                : "flex-1 bg-slate-200 text-slate-700 p-3 rounded-lg font-bold";

            document.getElementById("tabAdminBtn").className = !isUser
                ? "flex-1 bg-blue-600 text-white p-3 rounded-lg font-bold"
                : "flex-1 bg-slate-200 text-slate-700 p-3 rounded-lg font-bold";
        }

        function toggleRegisterMode() {
            isRegistering = !isRegistering;
            generatedOtp = null;
            isOtpVerified = false;

            document.getElementById("userFormTitle").textContent =
                isRegistering ? "Create Staff Account" : "Staff Sign In";

            document.getElementById("toggleRegisterBtn").textContent =
                isRegistering ? "Back to Sign In" : "Create Account";

            document.getElementById("userAuthSubmitBtn").textContent =
                isRegistering ? "Complete Registration" : "Sign In";

            document.getElementById("registrationFields").classList.toggle("hidden", !isRegistering);
            document.getElementById("otpSection").classList.add("hidden");
            document.getElementById("otpVerifyBadge").classList.add("hidden");
        }

        // DEMO OTP
        function sendOtp() {
            const phone = document.getElementById("regPhone").value.trim();

            if (!/^\d{10}$/.test(phone)) {
                showToast("Enter a valid 10-digit mobile number.", "error");
                return;
            }

            generatedOtp = String(Math.floor(1000 + Math.random() * 9000));
            isOtpVerified = false;

            document.getElementById("otpSection").classList.remove("hidden");
            document.getElementById("otpVerifyBadge").classList.add("hidden");

            alert("DEMO OTP: " + generatedOtp + "\nThis demo does not send a real SMS.");
        }

        function verifyOtp() {
            const entered = document.getElementById("regOtpInput").value.trim();

            if (generatedOtp && entered === generatedOtp) {
                isOtpVerified = true;
                document.getElementById("otpVerifyBadge").classList.remove("hidden");
                showToast("OTP verified successfully.", "success");
            } else {
                showToast("Incorrect OTP.", "error");
            }
        }

        // ADMIN LOGIN
        function handleAdminAuth(event) {
            event.preventDefault();

            const id = document.getElementById("adminInputId").value.trim();
            const password = document.getElementById("adminInputPass").value;

            if (
                id === ADMIN_CREDENTIALS.username &&
                password === ADMIN_CREDENTIALS.password
            ) {
                currentUser = {
                    id,
                    name: "Admin Manager",
                    role: "Administrator",
                    isAdmin: true
                };

                sessionStorage.setItem("ry_session", JSON.stringify(currentUser));

                updateNav();
                closeAuthModal();
                showAdminDashboard();
                showToast("Admin login successful.", "success");
            } else {
                showToast("Invalid admin credentials.", "error");
            }
        }

        // EMPLOYEE LOGIN / REGISTRATION
        function handleUserAuth(event) {
            event.preventDefault();

            const users = getUsers();
            const id = document.getElementById("userInputId").value.trim();
            const password = document.getElementById("userInputPass").value;

            if (!id || !password) {
                showToast("Enter your user ID and password.", "error");
                return;
            }

            if (isRegistering) {
                const name = document.getElementById("regFullName").value.trim();
                const phone = document.getElementById("regPhone").value.trim();
                const role = document.getElementById("regRole").value.trim() || "Team Member";

                if (!name || !phone) {
                    showToast("Enter your name and mobile number.", "error");
                    return;
                }

                if (!isOtpVerified) {
                    showToast("Verify the demo OTP before registering.", "error");
                    return;
                }

                if (users.some(user => user.id.toLowerCase() === id.toLowerCase())) {
                    showToast("This user ID is already registered.", "error");
                    return;
                }

                const newUser = {
                    id,
                    password,
                    name,
                    phone,
                    role,
                    dept: "General",
                    email: id.toLowerCase() + "@rycreations.com",
                    bio: "RY Creations team member",
                    isAdmin: false
                };

                users.push(newUser);
                saveUsers(users);

                currentUser = newUser;
                sessionStorage.setItem("ry_session", JSON.stringify(currentUser));

                updateNav();
                closeAuthModal();
                showUserDashboard();
                showToast("Account created successfully.", "success");

            } else {
                const user = users.find(item =>
                    item.id.toLowerCase() === id.toLowerCase() &&
                    item.password === password
                );

                if (!user) {
                    showToast("Invalid login or account not found.", "error");
                    return;
                }

                currentUser = { ...user, isAdmin: false };
                sessionStorage.setItem("ry_session", JSON.stringify(currentUser));

                updateNav();
                closeAuthModal();
                showUserDashboard();
                showToast("Welcome back, " + currentUser.name + "!", "success");
            }
        }

        // EMPLOYEE PROFILE
        function renderUserProfile() {
            if (!currentUser || currentUser.isAdmin) return;

            document.getElementById("userDashName").textContent = currentUser.name;
            document.getElementById("userDashRole").textContent = currentUser.role || "Team Member";
            document.getElementById("userDashId").textContent = "ID: " + currentUser.id;
            document.getElementById("userDashEmail").textContent = currentUser.email || "";
            document.getElementById("userDashDept").textContent = currentUser.dept || "General";
        }

        // SUBMIT WORK REPORT
        function handleProgressSubmit(event) {
            event.preventDefault();

            if (!currentUser || currentUser.isAdmin) return;

            const project = document.getElementById("logProject").value.trim();
            const hours = Number(document.getElementById("logHours").value);
            const details = document.getElementById("logDetails").value.trim();

            if (!project || !details || hours <= 0 || hours > 24) {
                showToast("Enter valid report information.", "error");
                return;
            }

            const logs = getLogs();

            logs.unshift({
                id: "LOG_" + Date.now(),
                userId: currentUser.id,
                userName: currentUser.name,
                project,
                hours,
                details,
                status: document.getElementById("logStatus").value,
                notes: document.getElementById("logNotes").value.trim(),
                date: new Date().toLocaleDateString(),
                reviewed: false
            });

            saveLogs(logs);
            document.getElementById("progressForm").reset();

            renderUserLogs();
            showToast("Progress report submitted.", "success");
        }

        // EMPLOYEE REPORT HISTORY
        function renderUserLogs() {
            if (!currentUser || currentUser.isAdmin) return;

            const logs = getLogs().filter(log => log.userId === currentUser.id);
            const container = document.getElementById("userLogHistory");

            document.getElementById("statTotalLogs").textContent = logs.length;
            document.getElementById("statApprovedLogs").textContent =
                logs.filter(log => log.reviewed).length;

            if (!logs.length) {
                container.innerHTML =
                    '<p class="text-slate-500 text-sm">No reports submitted yet.</p>';
                return;
            }

            container.innerHTML = logs.map(log => `
                <div class="border rounded-xl p-4 bg-white">
                    <div class="flex flex-wrap justify-between gap-2">
                        <h3 class="font-bold">${escapeHTML(log.project)}</h3>
                        <span class="text-xs font-bold ${statusColor(log.status)}">${escapeHTML(log.status)}</span>
                    </div>
                    <p class="text-sm text-slate-600 mt-2">${escapeHTML(log.details)}</p>
                    <p class="text-xs text-slate-500 mt-3">
                        ${escapeHTML(log.date)} · ${log.hours} hours
                    </p>
                    ${log.reviewed ? '<p class="text-emerald-600 text-xs font-bold mt-2">Reviewed by admin</p>' : ''}
                </div>
            `).join("");
        }

        // ADMIN REPORT TABLE
        function renderAdminReports() {
            const logs = getLogs();
            const users = getUsers();
            const search = document.getElementById("adminSearch").value.toLowerCase();

            document.getElementById("adminStatStaff").textContent = users.length;
            document.getElementById("adminStatLogs").textContent = logs.length;

            const filtered = logs.filter(log =>
                log.userName.toLowerCase().includes(search) ||
                log.project.toLowerCase().includes(search) ||
                log.details.toLowerCase().includes(search)
            );

            const tbody = document.getElementById("adminReportsTable");

            if (!filtered.length) {
                tbody.innerHTML = `
                    <tr>
                        <td colspan="6" class="p-6 text-center text-slate-500">
                            No reports found.
                        </td>
                    </tr>`;
                return;
            }

            tbody.innerHTML = filtered.map(log => `
                <tr class="border-b hover:bg-slate-50">
                    <td class="p-3 text-sm">${escapeHTML(log.date)}</td>
                    <td class="p-3 text-sm font-semibold">${escapeHTML(log.userName)}</td>
                    <td class="p-3 text-sm">
                        <p class="font-bold">${escapeHTML(log.project)}</p>
                        <p class="text-xs text-slate-500">${escapeHTML(log.details)}</p>
                    </td>
                    <td class="p-3 text-sm">${log.hours}</td>
                    <td class="p-3 text-xs font-bold ${statusColor(log.status)}">
                        ${escapeHTML(log.status)}
                        ${log.reviewed ? '<p class="text-emerald-600">Reviewed</p>' : ''}
                    </td>
                    <td class="p-3">
                        <button onclick="reviewLog('${log.id}')"
                            class="bg-emerald-600 hover:bg-emerald-700 text-white px-3 py-2 rounded-lg text-xs font-bold">
                            ${log.reviewed ? "Reviewed" : "Review"}
                        </button>
                    </td>
                </tr>
            `).join("");
        }

        // ADMIN REVIEW ACTION
        function reviewLog(logId) {
            if (!currentUser || !currentUser.isAdmin) {
                showToast("Admin access required.", "error");
                return;
            }

            const logs = getLogs();
            const log = logs.find(item => item.id === logId);

            if (!log) return;

            log.reviewed = true;
            saveLogs(logs);

            renderAdminReports();
            showToast("Report marked as reviewed.", "success");
        }

        // REGISTERED EMPLOYEES
        function renderAdminStaff() {
            const users = getUsers();
            const container = document.getElementById("adminStaffView");

            if (!users.length) {
                container.innerHTML =
                    '<p class="text-slate-500">No registered employees yet.</p>';
                return;
            }

            container.innerHTML = users.map(user => `
                <div class="bg-white border rounded-xl p-5">
                    <div class="flex items-center gap-3 mb-3">
                        <div class="w-12 h-12 bg-blue-600 text-white rounded-full flex items-center justify-center">
                            <i class="fa-solid fa-user"></i>
                        </div>
                        <div>
                            <h3 class="font-bold">${escapeHTML(user.name)}</h3>
                            <p class="text-xs text-blue-600">${escapeHTML(user.role || "Team Member")}</p>
                        </div>
                    </div>
                    <p class="text-sm text-slate-600">User ID: ${escapeHTML(user.id)}</p>
                    <p class="text-sm text-slate-600">Department: ${escapeHTML(user.dept || "General")}</p>
                    <p class="text-sm text-slate-600">Email: ${escapeHTML(user.email || "")}</p>
                </div>
            `).join("");
        }

        // UTILITIES
        function statusColor(status) {
            if (status === "Completed") return "text-emerald-600";
            if (status === "In Progress") return "text-blue-600";
            return "text-amber-600";
        }

        function escapeHTML(value) {
            return String(value ?? "").replace(/[&<>"']/g, character => ({
                "&": "&amp;",
                "<": "&lt;",
                ">": "&gt;",
                '"': "&quot;",
                "'": "&#39;"
            })[character]);
        }
    </script>
